cmake_minimum_required(VERSION 3.20)
project(fix_gateway VERSION 1.0 LANGUAGES C CXX)
find_package(SQLite3 REQUIRED)
set(qf "${CMAKE_BINARY_DIR}/deps/quickfix-1.16.0")
if(NOT EXISTS "${qf}/CMakeLists.txt")
  message(FATAL_ERROR "Run scripts/setup to fetch the pinned QuickFIX engine.")
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(QUICKFIX_EXAMPLES OFF CACHE BOOL "")
set(QUICKFIX_TESTS OFF CACHE BOOL "")
set(QUICKFIX_SHARED_LIBS OFF CACHE BOOL "")
set(QUICKFIX_LIB_OUTPUT_DIR "${CMAKE_BINARY_DIR}/lib" CACHE PATH "")
add_subdirectory("${qf}" quickfix EXCLUDE_FROM_ALL)
add_library(gateway_core src/domain.cpp src/journal.cpp src/session.cpp)
target_compile_features(gateway_core PUBLIC cxx_std_20)
target_include_directories(gateway_core PUBLIC src)
target_link_libraries(gateway_core PUBLIC quickfix SQLite::SQLite3)
add_dependencies(gateway_core QUICKFIX_HEADERS_COPY)
configure_file("${qf}/spec/FIX44.xml" config/FIX44.xml COPYONLY)
configure_file(config/simulator-v1.cfg config/simulator-v1.cfg COPYONLY)
target_compile_definitions(gateway_core PRIVATE GATEWAY_CONFIG_DIR="${CMAKE_BINARY_DIR}/config")
add_executable(fix-gateway src/main.cpp)
target_link_libraries(fix-gateway PRIVATE gateway_core)
include(CTest)
add_executable(conformance tests/conformance.cpp)
target_link_libraries(conformance PRIVATE gateway_core)
add_test(NAME conformance COMMAND conformance)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
add_test(NAME cli_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py"
  $<TARGET_FILE:fix-gateway> "${CMAKE_CURRENT_SOURCE_DIR}/data/fixtures/orders-v1.txt")
set_tests_properties(conformance cli_contract PROPERTIES TIMEOUT 45)
